// 1. Apertura y creación de Store con índices
const req = indexedDB.open('PortafolioPWA_DB', 1);
req.onupgradeneeded = (e) => {
  const store = e.target.result.createObjectStore('registros', { keyPath: 'id', autoIncrement: true });
  store.createIndex('by_titulo', 'titulo');
};

// 2. Transacción asíncrona de inserción
const tx = db.transaction(['registros'], 'readwrite');
tx.objectStore('registros').add({ titulo, categoria, descripcion, fecha });

// 3. Consulta de todos los registros
const records = await store.getAll();
