Código Técnico Implementado

Network-First Flow 
async function fetchNetworkFirst(apiUrl) {
  try {
    // 1. Intento primario por Red
    const res = await fetch(apiUrl);
    const json = await res.json();
    
    // 2. Guardar en IndexedDB local
    await apiCacheDB.saveApiResponse('last_api', json, apiUrl);
    return { data: json, source: 'network' };
  } catch (err) {
    // 3. Fallback automático a IndexedDB
    const cached = await apiCacheDB.getApiResponse('last_api');
    if (cached) return { data: cached.data, source: 'indexeddb' };
    throw err;
  }
}
